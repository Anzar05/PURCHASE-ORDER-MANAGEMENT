* SELECTION SCREEN VALIDATION
*---------------------------------------------------------------------*

AT SELECTION-SCREEN.


*---------------------------------------------------------------------*
* VALIDATE PO
*---------------------------------------------------------------------*

  IF S_EBELN-LOW IS NOT INITIAL.

    SELECT SINGLE EBELN
      INTO LV_EBELN
      FROM EKKO
      WHERE EBELN = S_EBELN-LOW.

    IF SY-SUBRC <> 0.

      MESSAGE 'Invalid Purchase Order Number'
        TYPE 'E'.

    ENDIF.

  ENDIF.


*---------------------------------------------------------------------*
* VALIDATE VENDOR
*---------------------------------------------------------------------*

  IF S_LIFNR-LOW IS NOT INITIAL.

    SELECT SINGLE LIFNR
      INTO LV_LIFNR
      FROM LFA1
      WHERE LIFNR = S_LIFNR-LOW.

    IF SY-SUBRC <> 0.

      MESSAGE 'Invalid Vendor Number'
        TYPE 'E'.

    ENDIF.

  ENDIF.


*---------------------------------------------------------------------*
* VALIDATE DATE RANGE
*---------------------------------------------------------------------*

  IF S_BEDAT-LOW IS NOT INITIAL
     AND S_BEDAT-HIGH IS NOT INITIAL.

    IF S_BEDAT-LOW > S_BEDAT-HIGH.

      MESSAGE 'From Date cannot be greater than To Date'
        TYPE 'E'.

    ENDIF.

  ENDIF.


*---------------------------------------------------------------------*
* START OF SELECTION
*---------------------------------------------------------------------*

START-OF-SELECTION.


*---------------------------------------------------------------------*
* GET DATA
*---------------------------------------------------------------------*

  PERFORM GET_PO_DATA.


*---------------------------------------------------------------------*
* NO DATA
*---------------------------------------------------------------------*

  IF GT_PO_DETAIL IS INITIAL.

    MESSAGE 'No Purchase Orders found for the selected criteria'
      TYPE 'I'.

    STOP.

  ENDIF.


*---------------------------------------------------------------------*
* MAIN REPORT
*---------------------------------------------------------------------*

  IF P_MAIN = 'X'.

    PERFORM BUILD_MAIN_RECORDS.

    PERFORM DISPLAY_MAIN_ALV.


*---------------------------------------------------------------------*
* VENDOR REPORT
*---------------------------------------------------------------------*

  ELSEIF P_VENDOR = 'X'.

    PERFORM BUILD_VENDOR_SUMMARY.

    PERFORM DISPLAY_VENDOR_SUMMARY.

  ENDIF.


*---------------------------------------------------------------------*
* FORM GET_PO_DATA
*---------------------------------------------------------------------*

FORM GET_PO_DATA.

  CLEAR GT_PO_DETAIL.


*---------------------------------------------------------------------*
* SELECT PURCHASE ORDER DATA
*---------------------------------------------------------------------*

  SELECT
         A~EBELN
         A~BEDAT
         A~LIFNR

         C~NAME1
         C~STRAS
         C~ORT01
         C~LAND1

         A~BUKRS
         A~EKORG
         A~EKGRP
         A~WAERS

         B~EBELP
         B~MATNR

         D~MAKTX

         B~MENGE
         B~MEINS

         B~NETPR
         B~PEINH

         B~WERKS
         B~LGORT

         B~ELIKZ

    INTO TABLE GT_PO_DETAIL

    FROM EKKO AS A

    INNER JOIN EKPO AS B
      ON A~EBELN = B~EBELN

    LEFT OUTER JOIN LFA1 AS C
      ON A~LIFNR = C~LIFNR

    LEFT OUTER JOIN MAKT AS D
      ON B~MATNR = D~MATNR
     AND D~SPRAS = SY-LANGU

    WHERE A~EBELN IN S_EBELN
      AND A~LIFNR IN S_LIFNR
      AND A~BEDAT IN S_BEDAT.


*---------------------------------------------------------------------*
* CHECK RESULT
*---------------------------------------------------------------------*

  IF SY-SUBRC <> 0
     OR GT_PO_DETAIL IS INITIAL.

    CLEAR GT_PO_DETAIL.

    RETURN.

  ENDIF.


*---------------------------------------------------------------------*
* CALCULATE TOTAL AND STATUS
*---------------------------------------------------------------------*

  LOOP AT GT_PO_DETAIL INTO GS_PO_DETAIL.


    IF GS_PO_DETAIL-PEINH IS INITIAL.

      GS_PO_DETAIL-TOTAL_AMOUNT =
          GS_PO_DETAIL-MENGE
        * GS_PO_DETAIL-NETPR.

    ELSE.

      GS_PO_DETAIL-TOTAL_AMOUNT =
          ( GS_PO_DETAIL-MENGE
          * GS_PO_DETAIL-NETPR )
          / GS_PO_DETAIL-PEINH.

    ENDIF.


*---------------------------------------------------------------------*
* DELIVERY STATUS
*---------------------------------------------------------------------*

    IF GS_PO_DETAIL-ELIKZ = 'X'.

      GS_PO_DETAIL-STATUS = 'Completed'.

    ELSE.

      GS_PO_DETAIL-STATUS = 'Open'.

    ENDIF.


    MODIFY GT_PO_DETAIL
      FROM GS_PO_DETAIL.

  ENDLOOP.

ENDFORM.


*---------------------------------------------------------------------*
* FORM BUILD_MAIN_RECORDS
*---------------------------------------------------------------------*

FORM BUILD_MAIN_RECORDS.

  CLEAR:
    GT_PO_MAIN,
    GT_TEMP_MAIN.


*---------------------------------------------------------------------*
* CREATE TEMPORARY RECORDS
*---------------------------------------------------------------------*

  LOOP AT GT_PO_DETAIL INTO GS_PO_DETAIL.

    CLEAR GS_TEMP_MAIN.


    GS_TEMP_MAIN-EBELN =
      GS_PO_DETAIL-EBELN.

    GS_TEMP_MAIN-BEDAT =
      GS_PO_DETAIL-BEDAT.

    GS_TEMP_MAIN-LIFNR =
      GS_PO_DETAIL-LIFNR.

    GS_TEMP_MAIN-NAME1 =
      GS_PO_DETAIL-NAME1.

    GS_TEMP_MAIN-WAERS =
      GS_PO_DETAIL-WAERS.

    GS_TEMP_MAIN-ITEM_COUNT =
      1.

    GS_TEMP_MAIN-TOTAL_AMOUNT =
      GS_PO_DETAIL-TOTAL_AMOUNT.

    GS_TEMP_MAIN-STATUS =
      GS_PO_DETAIL-STATUS.


    APPEND GS_TEMP_MAIN
      TO GT_TEMP_MAIN.

  ENDLOOP.


*---------------------------------------------------------------------*
* SORT
*---------------------------------------------------------------------*

  SORT GT_TEMP_MAIN BY EBELN.


*---------------------------------------------------------------------*
* CREATE ONE RECORD PER PO
*---------------------------------------------------------------------*

  LOOP AT GT_TEMP_MAIN INTO GS_TEMP_MAIN.

    READ TABLE GT_PO_MAIN
      INTO GS_PO_MAIN
      WITH KEY EBELN = GS_TEMP_MAIN-EBELN.


    IF SY-SUBRC <> 0.

      APPEND GS_TEMP_MAIN
        TO GT_PO_MAIN.

    ELSE.


      GS_PO_MAIN-ITEM_COUNT =
          GS_PO_MAIN-ITEM_COUNT
        + 1.


      GS_PO_MAIN-TOTAL_AMOUNT =
          GS_PO_MAIN-TOTAL_AMOUNT
        + GS_TEMP_MAIN-TOTAL_AMOUNT.


*---------------------------------------------------------------------*
* OPEN STATUS HAS PRIORITY
*---------------------------------------------------------------------*

      IF GS_TEMP_MAIN-STATUS = 'Open'.

        GS_PO_MAIN-STATUS = 'Open'.

      ENDIF.


      MODIFY GT_PO_MAIN
        FROM GS_PO_MAIN
        INDEX SY-TABIX.

    ENDIF.

  ENDLOOP.


*---------------------------------------------------------------------*
* SORT MAIN TABLE
*---------------------------------------------------------------------*

  SORT GT_PO_MAIN BY EBELN.

ENDFORM.


*---------------------------------------------------------------------*
* FORM BUILD_VENDOR_SUMMARY
*---------------------------------------------------------------------*

FORM BUILD_VENDOR_SUMMARY.

  DATA: LV_CURRENT_VENDOR
          TYPE LFA1-LIFNR.

  DATA: LV_CURRENT_CURRENCY
          TYPE EKKO-WAERS.

  DATA: LV_CURRENT_PO
          TYPE EKKO-EBELN.


  CLEAR:
    GT_VENDOR_SUMMARY,
    GS_VENDOR_SUMMARY,
    LV_CURRENT_VENDOR,
    LV_CURRENT_CURRENCY,
    LV_CURRENT_PO.


*---------------------------------------------------------------------*
* SORT
*---------------------------------------------------------------------*

  SORT GT_PO_DETAIL
    BY LIFNR WAERS EBELN EBELP.


*---------------------------------------------------------------------*
* BUILD SUMMARY
*---------------------------------------------------------------------*

  LOOP AT GT_PO_DETAIL INTO GS_PO_DETAIL.


    IF LV_CURRENT_VENDOR IS INITIAL
       OR LV_CURRENT_VENDOR <> GS_PO_DETAIL-LIFNR
       OR LV_CURRENT_CURRENCY <> GS_PO_DETAIL-WAERS.


*---------------------------------------------------------------------*
* APPEND PREVIOUS GROUP
*---------------------------------------------------------------------*

      IF LV_CURRENT_VENDOR IS NOT INITIAL.

        APPEND GS_VENDOR_SUMMARY
          TO GT_VENDOR_SUMMARY.

      ENDIF.


      CLEAR GS_VENDOR_SUMMARY.


*---------------------------------------------------------------------*
* VENDOR
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-LIFNR =
        GS_PO_DETAIL-LIFNR.


*---------------------------------------------------------------------*
* NAME
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-NAME1 =
        GS_PO_DETAIL-NAME1.


*---------------------------------------------------------------------*
* CURRENCY
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-WAERS =
        GS_PO_DETAIL-WAERS.


*---------------------------------------------------------------------*
* FIRST PO
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-PO_COUNT =
        1.


*---------------------------------------------------------------------*
* FIRST ITEM
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-ITEM_COUNT =
        1.


*---------------------------------------------------------------------*
* FIRST TOTAL
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-TOTAL_AMOUNT =
        GS_PO_DETAIL-TOTAL_AMOUNT.


      LV_CURRENT_VENDOR =
        GS_PO_DETAIL-LIFNR.


      LV_CURRENT_CURRENCY =
        GS_PO_DETAIL-WAERS.


      LV_CURRENT_PO =
        GS_PO_DETAIL-EBELN.


    ELSE.


*---------------------------------------------------------------------*
* ADD ITEM
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-ITEM_COUNT =
          GS_VENDOR_SUMMARY-ITEM_COUNT
        + 1.


*---------------------------------------------------------------------*
* ADD TOTAL
*---------------------------------------------------------------------*

      GS_VENDOR_SUMMARY-TOTAL_AMOUNT =
          GS_VENDOR_SUMMARY-TOTAL_AMOUNT
        + GS_PO_DETAIL-TOTAL_AMOUNT.


*---------------------------------------------------------------------*
* NEW PO
*---------------------------------------------------------------------*

      IF LV_CURRENT_PO <> GS_PO_DETAIL-EBELN.

        GS_VENDOR_SUMMARY-PO_COUNT =
            GS_VENDOR_SUMMARY-PO_COUNT
          + 1.


        LV_CURRENT_PO =
          GS_PO_DETAIL-EBELN.

      ENDIF.

    ENDIF.

  ENDLOOP.


*---------------------------------------------------------------------*
* APPEND LAST GROUP
*---------------------------------------------------------------------*

  IF LV_CURRENT_VENDOR IS NOT INITIAL.

    APPEND GS_VENDOR_SUMMARY
      TO GT_VENDOR_SUMMARY.

  ENDIF.


  SORT GT_VENDOR_SUMMARY
    BY LIFNR WAERS.

ENDFORM.

*---------------------------------------------------------------------*
* FORM DISPLAY_DETAIL
*---------------------------------------------------------------------*

FORM DISPLAY_DETAIL.

  DATA: LX_MSG_DETAIL
          TYPE REF TO CX_SALV_MSG.

  DATA: LX_NOT_FOUND_DETAIL
          TYPE REF TO CX_SALV_NOT_FOUND.

  DATA: LV_ITEM_COUNT
          TYPE I.

  DATA: LV_TOTAL_VALUE
          TYPE P DECIMALS 2.

  DATA: GO_DISPLAY_DETAIL
          TYPE REF TO CL_SALV_DISPLAY_SETTINGS.


*---------------------------------------------------------------------*
* CLEAR TABLE
*---------------------------------------------------------------------*

  CLEAR GT_PO_DETAIL.


*---------------------------------------------------------------------*
* GET SELECTED PO
*---------------------------------------------------------------------*

  SELECT
         A~EBELN
         A~BEDAT

         A~LIFNR

         C~NAME1
         C~STRAS
         C~ORT01
         C~LAND1

         A~BUKRS
         A~EKORG
         A~EKGRP
         A~WAERS

         B~EBELP
         B~MATNR

         D~MAKTX

         B~MENGE
         B~MEINS

         B~NETPR
         B~PEINH

         B~WERKS
         B~LGORT

         B~ELIKZ

    INTO TABLE GT_PO_DETAIL

    FROM EKKO AS A

    INNER JOIN EKPO AS B
      ON A~EBELN = B~EBELN

    LEFT OUTER JOIN LFA1 AS C
      ON A~LIFNR = C~LIFNR

    LEFT OUTER JOIN MAKT AS D
      ON B~MATNR = D~MATNR
     AND D~SPRAS = SY-LANGU

    WHERE A~EBELN = GV_SELECTED_EBELN.


*---------------------------------------------------------------------*
* CHECK DATA
*---------------------------------------------------------------------*

  IF GT_PO_DETAIL IS INITIAL.

    MESSAGE 'No details found for selected Purchase Order'
      TYPE 'I'.

    RETURN.

  ENDIF.


*---------------------------------------------------------------------*
* INITIALIZE
*---------------------------------------------------------------------*

  CLEAR:
    LV_ITEM_COUNT,
    LV_TOTAL_VALUE.


*---------------------------------------------------------------------*
* CALCULATE
*---------------------------------------------------------------------*

  LOOP AT GT_PO_DETAIL INTO GS_PO_DETAIL.


    IF GS_PO_DETAIL-PEINH IS INITIAL.

      GS_PO_DETAIL-TOTAL_AMOUNT =
          GS_PO_DETAIL-MENGE
        * GS_PO_DETAIL-NETPR.

    ELSE.

      GS_PO_DETAIL-TOTAL_AMOUNT =
          ( GS_PO_DETAIL-MENGE
          * GS_PO_DETAIL-NETPR )
          / GS_PO_DETAIL-PEINH.

    ENDIF.


*---------------------------------------------------------------------*
* STATUS
*---------------------------------------------------------------------*

    IF GS_PO_DETAIL-ELIKZ = 'X'.

      GS_PO_DETAIL-STATUS = 'Completed'.

    ELSE.

      GS_PO_DETAIL-STATUS = 'Open'.

    ENDIF.


    MODIFY GT_PO_DETAIL
      FROM GS_PO_DETAIL.


    LV_ITEM_COUNT =
        LV_ITEM_COUNT
      + 1.


    LV_TOTAL_VALUE =
        LV_TOTAL_VALUE
      + GS_PO_DETAIL-TOTAL_AMOUNT.

  ENDLOOP.


*---------------------------------------------------------------------*
* READ HEADER RECORD
*---------------------------------------------------------------------*

  READ TABLE GT_PO_DETAIL
    INTO GS_PO_DETAIL
    INDEX 1.


*---------------------------------------------------------------------*
* PO DETAIL HEADER
*---------------------------------------------------------------------*

  WRITE: / '============================================================'.
  WRITE: / '                    PURCHASE ORDER DETAILS'.
  WRITE: / '============================================================'.

  WRITE: / 'Company Code        : ', GS_PO_DETAIL-BUKRS.
  WRITE: / 'Purchasing Org.     : ', GS_PO_DETAIL-EKORG.
  WRITE: / 'Purchasing Group    : ', GS_PO_DETAIL-EKGRP.

  WRITE: / 'Vendor Address      : ', GS_PO_DETAIL-STRAS.
  WRITE: / 'City                : ', GS_PO_DETAIL-ORT01.
  WRITE: / 'Country             : ', GS_PO_DETAIL-LAND1.

  WRITE: / 'Currency            : ', GS_PO_DETAIL-WAERS.

  WRITE: / 'Total Items         : ', LV_ITEM_COUNT.
  WRITE: / 'PO Total            : ', LV_TOTAL_VALUE,
                                      GS_PO_DETAIL-WAERS.

  WRITE: / '============================================================'.
  WRITE: / '                         ITEM DETAILS'.
  WRITE: / '============================================================'.

  SKIP.


*---------------------------------------------------------------------*
* CREATE DETAIL ALV
*---------------------------------------------------------------------*

  TRY.


      CALL METHOD CL_SALV_TABLE=>FACTORY

        IMPORTING

          T_TABLE = GT_PO_DETAIL.

*---------------------------------------------------------------------*

        EXPORTING

* COLUMNS
